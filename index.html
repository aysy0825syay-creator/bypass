Delta Bypass

キーを入力してbypassリンクを生成

// server.js const express = require("express"); const cors = require("cors"); const crypto = require("crypto"); const app = express(); app.use(cors()); app.use(express.json()); // 有効なキーリスト(本来はDB) const VALID_KEYS = new Set([ "DELTA-TEST-1234", "DELTA-VIP-5678" ]); // 使用済みキー const USED_KEYS = new Set(); // トークン発行 function generateToken(key) { const raw = `${key}-${Date.now()}-${crypto.randomBytes(8).toString("hex")}`; return crypto.createHash("sha256").update(raw).digest("hex").slice(0, 32); } app.post("/api/bypass", (req, res) => { const { key } = req.body; if (!key) { return res.json({ success: false, message: "キーが必要です" }); } if (!VALID_KEYS.has(key)) { return res.json({ success: false, message: "無効なキーです" }); } if (USED_KEYS.has(key)) { return res.json({ success: false, message: "このキーは使用済みです" }); } // キー消費 USED_KEYS.add(key); const token = generateToken(key); const link = `https://your-bypass.example.com/getkey?token=${token}`; res.json({ success: true, link }); }); app.listen(3000, () => console.log("Server running on http://localhost:3000")); npm init -y npm install express cors node server.js
